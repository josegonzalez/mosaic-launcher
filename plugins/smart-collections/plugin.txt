id=smart-collections
name=Smart Collections
api=1
category=Library
summary=Sorts your games into collections by franchise, play history, genre and decade.
description=Sorts your games into collections without you building any lists: franchises such as Mario and Zelda, games never or recently played, genres, decades and rules of your own.
# the smart:<name> entry sets
provides=sets.smart
# the franchises that ship with the plugin, such as Mario, Zelda and Sonic
setting.builtin=toggle|Built-in franchises|on
# more franchises, comma-separated, eg. Kirby, Bomberman
setting.extra=text|More franchises|
# the fewest games a collection needs to show
setting.min=choice|Smallest collection|2|1,2,3,5,10
# a collection of the games never played
setting.never=toggle|Never played|on
# a collection of the games played in the last recent_days days
setting.recent=toggle|Recently played|on
setting.recent_days=choice|Recently played (days)|7|1,3,7,14,30
# a collection of the games played over long_hours hours in all
setting.long=toggle|Long played|on
setting.long_hours=choice|Long played (hours)|10|1,5,10,20,50
# a collection of the games put on the card in the last added_days days
setting.added=toggle|Recently added|on
setting.added_days=choice|Recently added (days)|30|7,30,90
# a collection for each decade the games' .info years fall in
setting.decades=toggle|By decade|on
# a collection for each genre the games' .info files name
setting.genres=toggle|By genre|on
