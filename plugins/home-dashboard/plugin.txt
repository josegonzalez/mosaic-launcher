slots=home
# the service first-run sets the mode through
provides=home-dashboard.setup
# loads after favorites, whose collection it reads as it starts
optional=favorites
# dashboard: tabs with a continue card, pinned games and stats; minui: a plain list of recent games;
# retroachievements: recent and pinned games with achievements; pico8: the card's Pico-8 carts;
# minimal: the continue card and the clock; consoles: every system as a tile with its game count
setting.mode=choice|Home mode|dashboard|dashboard,minui,retroachievements,pico8,minimal,consoles
# the rows under the continue card
setting.shelf=toggle|Top shelf|on
# the rows under the continue card, in order: continue, recent, added, played, random, shortcuts and missions
setting.rows=text|Home rows|continue, recent
# the actions the shortcuts row runs, by action or plugin id
setting.shortcuts=text|Shortcuts|search, random-game, activity-stats, system-info
# the minimal mode's clock
setting.clock=choice|Minimal clock|24h|24h,12h
# favorites' games, while favorites is on
setting.pinned=toggle|Pinned tab|on
# play time from stock's play tracker
setting.stats=toggle|Stats tab|on
# the tags of the folders the pico8 mode lists carts from, eg. Roms/Pico-8 (P8)
setting.pico8_tags=text|Pico-8 folder tags|P8,PICO8,PICO
