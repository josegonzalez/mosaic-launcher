id=romm
name=RomM
api=1
category=Network
summary=Browses a RomM server's library, downloads games and backs up saves.
# the RomM server, eg. http://192.168.1.20:8080 or https://romm.home; http:// sends the login in clear text
setting.url=text|Server|||protected
# RomM user name; leave it empty to log in with an API token instead
setting.user=text|User|||protected
# the user's password, or a RomM API token when User is empty
setting.password=text|Password or API token|||protected,secret
# https:// only: the server certificate's SHA-256 fingerprint, for a self-signed home server
setting.pin=text|Certificate fingerprint (SHA-256)|||protected
# https:// only: a PEM file of CA certificates to trust in place of the device's own
setting.ca_file=text|CA certificates file|||protected
# upload a game's save after it returns, when it changed
setting.sync=toggle|Upload saves after playing|on
# how many of a game's uploaded saves the server keeps
setting.keep=choice|Saves kept on the server|10|5,10,20,50
# the game list button that opens the selected game's Flashback, shared with other plugins' actions on it
setting.button=choice|Flashback button|Y|none,Y,L2,R2,SELECT|button:romm.flashback
