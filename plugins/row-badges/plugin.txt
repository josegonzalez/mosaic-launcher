description=Adds small dots after a game's name for games you haven't played, your favorites, games with a tag or a manual, and tiers, and icons for a game's status and for 100% of its achievements.
# the games.played service: whether a game was played, the one definition of it
provides=games.played
# a dot on games that weren't played: not in recents, with no save, state or resume slot, and no session in the play history
setting.unplayed=toggle|Unplayed dot|off||public
# dots for the tier= value, 1 to 5, in a game's or system's .info sidecar
setting.tiers=toggle|Tier dots|off||public
# a dot on your favorites, from the favorites plugin
setting.favorites=toggle|Favorite mark|off
# a dot on games with any tag, from the tags plugin
setting.tagged=toggle|Tag mark|off
# a dot on games with a manual, from the manuals plugin
setting.manual=toggle|Manual mark|off
# a star on games tags marks Completed, from its Status menu
setting.completed=toggle|Completed icon|off
# a tick on games tags marks Beaten
setting.beaten=toggle|Beaten icon|off
# a cross on games tags marks Broken
setting.broken=toggle|Broken icon|off
# an hourglass on games tags marks Backlog
setting.backlog=toggle|Backlog icon|off
# a trophy on games with every RetroAchievement unlocked, from minarch's local progress
setting.perfect=toggle|100% achievements trophy|off
