setting.rclone_config=text|rclone config file|||protected

setting.rclone_pass=text|rclone config password|||protected,secret
# rsync: user@host of the other device, or empty for a folder on this one
setting.host=text|rsync host|||protected
# rsync: the other device's SSH port
setting.port=text|rsync SSH port|22||protected
# rsync: a private key file with no passphrase; empty for .userdata/<platform>/launcher/sync/ssh_key
setting.key=text|rsync SSH key file|||protected
# rsync: the folder to sync with on the other device
setting.path=text|rsync folder|||protected
# Syncthing: the other device's ID, from its Syncthing
setting.peer=text|Syncthing device ID|||protected
# Syncthing: the key for its API on 127.0.0.1, made when it's empty
setting.api_key=text|Syncthing API key|||protected,secret
