id=theme-plus
name=Theme plus
api=1
category=Looks
summary=Changes colours, fonts, text size, wallpapers and system icons from a theme.
description=Changes how the launcher looks, its colours, font, text size, wallpapers, system icons and button hints, from a theme, or one that changes each start or each day, without touching stock's settings.
slots=theme
# the service first-run sets the theme through
provides=theme-plus.setup
# the theme: a folder name in Themes, a bundled theme, a palette's name, or stock for no change
setting.theme=text|Theme|stock
# a Theme tile on the quick menu that lists the themes
setting.tile=toggle|Quick menu tile|on
# a game list button that lists the themes
setting.button=choice|Theme button|none|none,Y,L2,R2,SELECT|button:theme-plus.pick
# a random theme each time the launcher starts (start) or one a day (daily) in place of the theme setting's
setting.rotate=choice|Rotate|off|off,start,daily
# the themes a rotation picks from, separated by commas; empty picks from every theme the picker lists
setting.rotate_between=text|Rotate between|
# a folder in Wallpapers with a <system tag>.png for each system's list, over the theme's own; empty for the theme's
setting.wallpaper_pack=text|Wallpaper pack|
# a folder in Icons with a <system tag>.png for each system, over the theme's own; empty for the theme's
setting.icon_pack=text|Icon pack|
# the rest override the theme's values; theme keeps the theme's
# the background colour as 0xRRGGBB
setting.background=text|Background colour|theme
setting.text_size=choice|Text size|theme|theme,small,stock,large,larger,largest
# the least contrast ratio text keeps
setting.contrast=choice|Contrast|theme|theme,off,3,4.5,7
setting.ui_size=choice|UI size|theme|theme,stock,compact,large,larger
setting.margins=choice|Screen margins|theme|theme,auto,2,5,8,10,15,20
# a font for the characters the UI font has no glyphs for, relative to the card or a full path; empty keeps the theme's
setting.fallback_font=text|Fallback font|
