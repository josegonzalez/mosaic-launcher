id=hold-lock
name=Hold lock
api=1
category=Controls and setup
summary=Locks every button so nothing changes by accident.
# held for the hold setting's seconds, locks every button, and again unlocks them; off for none.
# A long power press still powers off, and the volume and brightness keys can't be locked.
setting.chord=choice|Lock chord|select+start|select+start,l1+r1,l2+r2,off
setting.hold=choice|Hold the chord for (seconds)|1|1,2,3
# the switch on the side of the tg5040 and tg5050 locks while it's on. It's stock's mute switch,
# so it still mutes as stock's Mute settings say. Unsupported on the h700 and desktop, which have none.
setting.switch=toggle|Lock with the hold switch|off
