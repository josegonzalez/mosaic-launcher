id=first-run
name=First run
api=1
category=Controls and setup
summary=A short setup wizard the first time the launcher starts.
description=A short setup wizard the first time the launcher starts, with a few settings, plugins to try and what each button does, and a help overlay of the buttons on any screen.
# set once the wizard is finished or skipped; turn it off to see it again at the next start
setting.seen=toggle|Already shown|off
# a First run tile on the quick menu that opens the wizard
setting.tile=toggle|Quick menu tile|on
# a game list button that opens the wizard
setting.button=choice|First run button|none|none,Y,L2,R2,SELECT|button:first-run.open
# the page of stock's clock, battery percentage, haptics and notification toggles
setting.toggles=toggle|Settings page|on
# the page of other plugins' settings: the home mode, the favorites collection, the theme and the language, for each plugin that's on
setting.setup=toggle|Plugin settings page|on
# the page saying whether the launcher has a PIN
setting.pin=toggle|PIN page|on
# plugin ids the plugins page suggests, comma-separated; empty leaves the page out
setting.suggest=text|Suggested plugins|home-dashboard,switcher-plus,favorites,search
# the page of what's on each button
setting.buttons=toggle|Buttons page|on
# a Buttons tile on the quick menu that opens the buttons page on its own
setting.buttons_tile=toggle|Buttons tile|on
# a game list button that opens the buttons page on its own
setting.buttons_button=choice|Buttons button|none|none,Y,L2,R2,SELECT|button:first-run.buttons
# a Help tile on the quick menu that shows what each button does on the screen under it
setting.help_tile=toggle|Help tile|on
# a game list button, or a button held for a second on any screen, that shows the help
setting.help_button=choice|Help button|none|none,Y,L2,R2,hold SELECT,hold START|button:first-run.help
